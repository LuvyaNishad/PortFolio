/* ─────────────────────────────────────────────────────────────────────────────
   BACKGROUND — animated gradient mesh + cursor trail
───────────────────────────────────────────────────────────────────────────── */
const Background = () => {
  const canvasRef = useRef(null);
  const target    = useRef({ x: -999, y: -999 });
  const trail     = useRef([]);
  const rafRef    = useRef(null);
  const HEAD      = useRef({ x: -999, y: -999 });

  // Trail colour — lerps between olive (light bg) and sage (dark bg)
  const colLight  = { r:65,  g:67,  b:27  };
  const colDark   = { r:174, g:183, b:132 };
  const trailCol  = useRef({ ...colLight }); // ← top-level, legal

  // Trail config
  const TRAIL_LENGTH = 28;
  const MAX_AGE      = 28;
  const BASE_RADIUS  = 7;
  const LERP         = 0.18;

  useEffect(() => {
    const onMove = e => {
      target.current = { x: e.clientX, y: e.clientY };
    };
    window.addEventListener("mousemove", onMove, { passive: true });

    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");

    const resize = () => {
      canvas.width  = window.innerWidth;
      canvas.height = window.innerHeight;
      if (HEAD.current.x < 0) {
        HEAD.current = { x: canvas.width / 2, y: canvas.height / 2 };
        target.current = { ...HEAD.current };
      }
    };
    resize();
    window.addEventListener("resize", resize, { passive: true });

    // Trail colour — lerped each frame, declared at component level above
    const draw = () => {
      const { width: W, height: H } = canvas;

      // Detect if cursor is over a dark section by checking element under pointer
      const el = document.elementFromPoint(target.current.x, target.current.y);
      const inDark = el?.closest?.("#contact, #contact *") !== null &&
                     el?.closest?.("#contact") !== null;

      // Lerp trail colour toward target palette (0.06 = smooth, ~1s transition)
      const targetCol = inDark ? colDark : colLight;
      trailCol.current.r += (targetCol.r - trailCol.current.r) * 0.06;
      trailCol.current.g += (targetCol.g - trailCol.current.g) * 0.06;
      trailCol.current.b += (targetCol.b - trailCol.current.b) * 0.06;
      const { r, g, b } = trailCol.current;

      // Lerp head toward real cursor
      HEAD.current.x += (target.current.x - HEAD.current.x) * LERP;
      HEAD.current.y += (target.current.y - HEAD.current.y) * LERP;

      // Spawn a new particle at head position every frame
      trail.current.push({
        x:    HEAD.current.x,
        y:    HEAD.current.y,
        age:  0,
        size: BASE_RADIUS,
        r, g, b,   // snapshot colour at birth
      });

      // Age all particles; remove dead ones
      trail.current = trail.current
        .map(p => ({ ...p, age: p.age + 1 }))
        .filter(p => p.age < MAX_AGE)
        .slice(-TRAIL_LENGTH);

      // Clear
      ctx.clearRect(0, 0, W, H);

      // Draw each particle — oldest first (back), newest last (front)
      trail.current.forEach(p => {
        const life    = 1 - p.age / MAX_AGE;
        const radius  = p.size * (0.3 + life * 0.7);
        const alpha   = life * life * 0.55;
        const blur    = (1 - life) * 12;
        const pr = p.r ?? 65, pg = p.g ?? 67, pb = p.b ?? 27;

        const grad = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, radius * 3.5);
        grad.addColorStop(0,   `rgba(${pr},${pg},${pb},${(alpha * 1.8).toFixed(3)})`);
        grad.addColorStop(0.4, `rgba(${pr},${pg},${pb},${(alpha * 0.9).toFixed(3)})`);
        grad.addColorStop(1,   `rgba(${pr},${pg},${pb},0)`);

        ctx.save();
        if (blur > 0.5) ctx.filter = `blur(${blur.toFixed(1)}px)`;
        ctx.fillStyle = grad;
        ctx.beginPath();
        ctx.arc(p.x, p.y, radius * 3.5, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
      });

      rafRef.current = requestAnimationFrame(draw);
    };

    rafRef.current = requestAnimationFrame(draw);

    return () => {
      window.removeEventListener("mousemove", onMove);
      window.removeEventListener("resize", resize);
      cancelAnimationFrame(rafRef.current);
    };
  }, []);

  return (
    <>
    {/* ── Canvas trail — fixed above ALL sections including dark contact ── */}
    <canvas ref={canvasRef} style={{
      position:"fixed", inset:0, width:"100%", height:"100%",
      zIndex:50,           // above sections (zIndex:1) below nav (zIndex:200)
      pointerEvents:"none",
      mixBlendMode:"normal",
    }}/>

    <div style={{ position:"fixed", inset:0, zIndex:0, overflow:"hidden", pointerEvents:"none" }}>

      {/* ── Layer 0: cream base — always solid ── */}
      <div style={{ position:"absolute", inset:0, background:"#F0EDD8" }}/>

      {/* ── Layer 1: large travelling gradient mesh ──────────────────────────
          A 300% × 300% linear gradient shifts position over 22s — because the
          gradient is much bigger than the viewport, position changes look like
          the colour field itself is flowing across the screen.
      ─────────────────────────────────────────────────────────────────────── */}
      <div style={{
        position:"absolute", inset:0,
        background:`linear-gradient(
          135deg,
          #F8F3E1 0%,
          rgba(174,183,132,0.80) 18%,
          rgba(248,243,225,0.90) 32%,
          rgba(227,219,187,0.85) 48%,
          rgba(174,183,132,0.70) 62%,
          rgba(248,243,225,0.95) 76%,
          rgba(227,219,187,0.80) 88%,
          #F0EDD8 100%
        )`,
        backgroundSize:"300% 300%",
        animation:"meshShift 22s ease-in-out infinite",
        mixBlendMode:"normal",
      }}/>

      {/* ── Layers 2–7: six independent fold blobs ───────────────────────────
          Each is a large elliptical radial gradient positioned at a different
          quadrant. They animate with translate + scale + rotate independently
          so they fold over and under each other — this is what creates the
          "fabric" / "silk" look. Opacities are intentionally high (0.60–0.75)
          so the movement is clearly visible without being garish.
      ─────────────────────────────────────────────────────────────────────── */}

      {/* Fold A — sage, top-left, slow drift */}
      <div style={{
        position:"absolute",
        width:"90vw", height:"80vw",
        top:"-20%", left:"-20%",
        background:"radial-gradient(ellipse 60% 55% at 45% 50%, rgba(174,183,132,0.72) 0%, transparent 70%)",
        filter:"blur(55px)",
        animation:"fold1 19s ease-in-out infinite",
      }}/>

      {/* Fold B — sand, bottom-right, medium */}
      <div style={{
        position:"absolute",
        width:"85vw", height:"90vw",
        top:"35%", left:"30%",
        background:"radial-gradient(ellipse 55% 60% at 50% 45%, rgba(227,219,187,0.78) 0%, transparent 70%)",
        filter:"blur(60px)",
        animation:"fold2 24s ease-in-out infinite",
      }}/>

      {/* Fold C — sage, centre, fast */}
      <div style={{
        position:"absolute",
        width:"70vw", height:"65vw",
        top:"15%", left:"25%",
        background:"radial-gradient(ellipse 58% 52% at 48% 52%, rgba(174,183,132,0.60) 0%, transparent 68%)",
        filter:"blur(48px)",
        animation:"fold3 15s ease-in-out infinite",
      }}/>

      {/* Fold D — sand accent, top-right, medium-fast */}
      <div style={{
        position:"absolute",
        width:"75vw", height:"70vw",
        top:"-10%", left:"45%",
        background:"radial-gradient(ellipse 52% 58% at 52% 48%, rgba(227,219,187,0.68) 0%, transparent 70%)",
        filter:"blur(52px)",
        animation:"fold4 20s ease-in-out infinite",
      }}/>

      {/* Fold E — deep sage, bottom-left, slowest */}
      <div style={{
        position:"absolute",
        width:"80vw", height:"75vw",
        top:"55%", left:"-15%",
        background:"radial-gradient(ellipse 56% 54% at 46% 54%, rgba(174,183,132,0.65) 0%, transparent 72%)",
        filter:"blur(58px)",
        animation:"fold5 28s ease-in-out infinite",
      }}/>

      {/* Fold F — warm sand, middle-right, medium */}
      <div style={{
        position:"absolute",
        width:"65vw", height:"60vw",
        top:"30%", left:"55%",
        background:"radial-gradient(ellipse 54% 56% at 50% 50%, rgba(227,219,187,0.70) 0%, transparent 68%)",
        filter:"blur(45px)",
        animation:"fold6 17s ease-in-out infinite",
      }}/>

      {/* ── Layer 8: central cream vignette — keeps text readable ── */}
      <div style={{
        position:"absolute", inset:0,
        background:`radial-gradient(ellipse 75% 65% at 50% 45%,
          rgba(248,243,225,0.50) 0%,
          transparent 75%
        )`,
      }}/>

      {/* ── Layer 5: grain ── */}
      <div style={{ position:"absolute", inset:0, opacity:0.030,
        backgroundImage:`url("data:image/svg+xml,%3Csvg viewBox='0 0 256 256' xmlns='http://www.w3.org/2000/svg'%3E%3Cfilter id='n'%3E%3CfeTurbulence type='fractalNoise' baseFrequency='0.9' numOctaves='4' stitchTiles='stitch'/%3E%3C/filter%3E%3Crect width='100%25' height='100%25' filter='url(%23n)'/%3E%3C/svg%3E")`}}/>
    </div>
    </>
  );
};

/* ─────────────────────────────────────────────────────────────────────────────
   LOADER
   Sequence:
     0–0.7s  → dark screen, logo letters stagger in from below
     0.7–1.4s → tagline line draws in from left, counter 0→100
     1.4–2.0s → progress bar completes
     2.0–2.8s → curtain splits: top half slides up, bottom slides down → reveals portfolio
───────────────────────────────────────────────────────────────────────────── */
const Loader = ({ onDone }) => {
  const [phase,   setPhase]   = useState(0); // 0=in 1=counter 2=split 3=done
  const [count,   setCount]   = useState(0);
  const letters = "AURELIUS".split("");

  useEffect(() => {
    const t1 = setTimeout(() => setPhase(1), 700);
    const t2 = setTimeout(() => setPhase(2), 1900);
    const t3 = setTimeout(() => { setPhase(3); onDone(); }, 2750);
    return () => [t1,t2,t3].forEach(clearTimeout);
  }, []);

  // Counter 0 → 100 during phase 1
  useEffect(() => {
    if (phase !== 1) return;
    let n = 0;
    const iv = setInterval(() => {
      n += Math.ceil(Math.random() * 6);
      if (n >= 100) { setCount(100); clearInterval(iv); }
      else setCount(n);

    return () => clearInterval(iv);
  }, [phase]);

  return (
    <AnimatePresence>

/* ─────────────────────────────────────────────────────────────────────────────
   REVEAL VARIANTS — different entrance animations per section
   Use: <Reveal variant="slideLeft|slideRight|zoom|stagger" ...>
───────────────────────────────────────────────────────────────────────────── */
const REVEAL_VARIANTS = {
  up:         { hidden:{opacity:0,y:40},     visible:{opacity:1,y:0} },
  slideLeft:  { hidden:{opacity:0,x:-60},    visible:{opacity:1,x:0} },
  slideRight: { hidden:{opacity:0,x:60},     visible:{opacity:1,x:0} },
  zoom:       { hidden:{opacity:0,scale:.88},visible:{opacity:1,scale:1} },
  fade:       { hidden:{opacity:0},          visible:{opacity:1} },
};

const Reveal = ({children, delay=0, y=32, variant="up", style={}}) => {
  const ref = useRef(null);
  const inV = useInView(ref,{once:true,margin:"-60px"});
  const v   = REVEAL_VARIANTS[variant] || REVEAL_VARIANTS.up;
  return (
    <motion.div ref={ref} style={style}
      initial={v.hidden}
      animate={inV ? v.visible : v.hidden}
      transition={{duration:.9, delay, ease:[.22,1,.36,1]}}>
      {children}
    </motion.div>
  );
};

/* ─────────────────────────────────────────────────────────────────────────────
   SLABEL — scrambles characters on hover, then resolves
───────────────────────────────────────────────────────────────────────────── */
const CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789✦";
const SLabel = ({children}) => {
  const [display, setDisplay] = useState(children);
  const rafRef = useRef(null);

  const scramble = () => {
    const target = String(children).toUpperCase();
    let iter = 0;
    cancelAnimationFrame(rafRef.current);
    const tick = () => {
      setDisplay(
        target.split("").map((ch, i) =>
          i < iter ? ch :
          ch === " " ? " " :
          CHARS[Math.floor(Math.random() * CHARS.length)]
        ).join("")
      );
      if (iter < target.length) { iter += 0.4; rafRef.current = requestAnimationFrame(tick); }
      else setDisplay(target);
    };
    rafRef.current = requestAnimationFrame(tick);
  };

  useEffect(() => () => cancelAnimationFrame(rafRef.current), []);

  return (
    <div
      onMouseEnter={scramble}
      style={{
        fontSize:11, textTransform:"uppercase", letterSpacing:"0.18em",
        marginBottom:14, fontWeight:500,
        cursor:"default", userSelect:"none",
        fontFamily:"'DM Sans',monospace",
        display:"flex", alignItems:"center", gap:6,
      }}>
      <span style={{color:"#41431B", fontSize:9}}>✦</span>
      <span style={{color:"#AEB784"}}>{display}</span>
    </div>
  );
};

/* ─────────────────────────────────────────────────────────────────────────────
   COUNT-UP — animates a number from 0 to target when it enters view
───────────────────────────────────────────────────────────────────────────── */
const CountUp = ({to, duration=1200, style={}}) => {
  const [val, setVal] = useState(0);
  const ref = useRef(null);
  const inV = useInView(ref, {once:true, margin:"-80px"});

  useEffect(() => {
    if (!inV) return;
    const start = performance.now();
    const tick = now => {
      const p = Math.min((now - start) / duration, 1);
      const ease = 1 - Math.pow(1 - p, 3); // cubic ease-out
      setVal(Math.round(ease * to));
      if (p < 1) requestAnimationFrame(tick);
    };
    requestAnimationFrame(tick);
  }, [inV]);

  return <span ref={ref} style={style}>{String(val).padStart(2,"0")}</span>;
};

/* ─────────────────────────────────────────────────────────────────────────────
   CUSTOM CURSOR — colour-aware
   Dot and ring lerp from olive → sage/cream when entering #contact / #footer
───────────────────────────────────────────────────────────────────────────── */
const CustomCursor = () => {
  const dot    = useRef(null);
  const ring   = useRef(null);
  const mouse  = useRef({x:0, y:0});
  const pos    = useRef({x:0, y:0});
  const raf    = useRef(null);
  // Lerped colour channels (olive by default)
  const col    = useRef({r:74, g:92, b:40});
  const [hov,     setHov]     = useState(false);
  const [visible, setVisible] = useState(false);

  useEffect(()=>{
    const onMove = e => {
      mouse.current = {x: e.clientX, y: e.clientY};
      if (!visible) setVisible(true);
    };
    const onLeave = () => setVisible(false);
    const onEnter = () => setVisible(true);

    const addHov = () => {
      document.querySelectorAll("a,button,[data-cursor]").forEach(el => {
        el.addEventListener("mouseenter", ()=>setHov(true));
        el.addEventListener("mouseleave", ()=>setHov(false));
      });
    };

    document.addEventListener("mousemove", onMove);
    document.addEventListener("mouseleave", onLeave);
    document.addEventListener("mouseenter", onEnter);
    addHov();

    // Light colour (sage) for dark sections, dark olive for light sections
    const DARK = {r:174, g:183, b:132};   // sage — visible on dark bg
    const LITE = {r:74,  g:92,  b:40 };   // olive — visible on light bg

    const loop = () => {
      // Detect if cursor is over the dark section
      const el = document.elementFromPoint(mouse.current.x, mouse.current.y);
      const inDark = !!el?.closest?.("#contact, footer");

      // Lerp colour
      const target = inDark ? DARK : LITE;
      col.current.r += (target.r - col.current.r) * 0.08;
      col.current.g += (target.g - col.current.g) * 0.08;
      col.current.b += (target.b - col.current.b) * 0.08;
      const {r, g, b} = col.current;
      const rgb = `${Math.round(r)},${Math.round(g)},${Math.round(b)}`;

      // Lerp ring position
      pos.current.x += (mouse.current.x - pos.current.x) * 0.12;
      pos.current.y += (mouse.current.y - pos.current.y) * 0.12;

      // Apply transforms + colours directly — no React re-render
      if (dot.current) {
        dot.current.style.transform = `translate(${mouse.current.x}px,${mouse.current.y}px) translate(-50%,-50%)`;
        dot.current.style.background = `rgb(${rgb})`;
      }
      if (ring.current) {
        ring.current.style.transform = `translate(${pos.current.x}px,${pos.current.y}px) translate(-50%,-50%)`;
        ring.current.style.borderColor = inDark
          ? `rgba(${rgb},0.65)`
          : `rgba(${rgb},0.45)`;
        ring.current.style.background = inDark
          ? `rgba(${rgb},0.08)`
          : "transparent";
      }

      raf.current = requestAnimationFrame(loop);
    };
    raf.current = requestAnimationFrame(loop);

    return () => {
      document.removeEventListener("mousemove", onMove);
      document.removeEventListener("mouseleave", onLeave);
      document.removeEventListener("mouseenter", onEnter);
      cancelAnimationFrame(raf.current);
    };
  }, []);

  return (
    <>
      <div ref={dot} style={{
        position:"fixed", top:0, left:0, zIndex:9999,
        width:6, height:6, borderRadius:"50%",
        background:"rgb(74,92,40)",
        pointerEvents:"none", willChange:"transform",
        opacity: visible ? 1 : 0,
        transition:"opacity .3s",
      }}/>
      <div ref={ring} style={{
        position:"fixed", top:0, left:0, zIndex:9998,
        width: hov ? 52 : 36,
        height: hov ? 52 : 36,
        borderRadius:"50%",
        border:"1.5px solid rgba(74,92,40,0.45)",
        background:"transparent",
        pointerEvents:"none", willChange:"transform",
        opacity: visible ? 1 : 0,
        transition:"opacity .3s, width .35s cubic-bezier(.22,1,.36,1), height .35s cubic-bezier(.22,1,.36,1)",
      }}/>
    </>
  );
};

/* ─────────────────────────────────────────────────────────────────────────────
   NAV — detached floating pill
   • Scroll-spy: IntersectionObserver watches each section, highlights active
   • Lamp effect: bright bar + radial glow above active item (from the ref)
   • Green stained-glass pill outline, detached from top edge
───────────────────────────────────────────────────────────────────────────── */
const NAV_LINKS = [
  {label:"About",    id:"about"},
  {label:"Tools",    id:"tools"},
  {label:"Library",  id:"library"},
  {label:"Contact",  id:"contact"},
  {label:"Resume",   id:"resume", external:"YOUR_RESUME_URL_HERE"},
];

const Nav = () => {
  const [active, setActive]  = useState("hero");
  const [lampX,  setLampX]   = useState(null);
  const pillRef   = useRef(null);
  const linkRefs  = useRef({});
  const navLocked = useRef(false);   // true while a manual click is in flight
  const lockTimer = useRef(null);

  const go = id => {
    // Lock the observer so it can't overwrite our click
    navLocked.current = true;
    clearTimeout(lockTimer.current);
    lockTimer.current = setTimeout(()=>{ navLocked.current = false; }, 900);

    setActive(id);
    document.getElementById(id)?.scrollIntoView({behavior:"smooth"});
  };

  // Scroll-spy — only fires when NOT locked by a manual click
  useEffect(()=>{
    const sections = ["hero","about","tools","library","thumbnails",
      "video-edits","graphic-design","code-projects","contact"];
    const obs = new IntersectionObserver(entries=>{
      if (navLocked.current) return;           // ← ignore while user clicked
      entries.forEach(en=>{ if(en.isIntersecting) setActive(en.target.id); });
    },{threshold:0.35});
    sections.forEach(id=>{ const el=document.getElementById(id); if(el) obs.observe(el); });
    return ()=>{ obs.disconnect(); clearTimeout(lockTimer.current); };
  },[]);

  // Map work sub-sections → "library" tab
  const displayActive = ["thumbnails","video-edits","graphic-design","code-projects"]
    .includes(active) ? "library" : active;

  // Measure active link centre relative to pill for lamp position
  useEffect(()=>{
    const linkEl = linkRefs.current[displayActive];
    const pillEl = pillRef.current;
    if (!linkEl || !pillEl) return;
    const lr = linkEl.getBoundingClientRect();
    const pr = pillEl.getBoundingClientRect();
    setLampX(lr.left - pr.left + lr.width / 2);
  }, [displayActive]);

  return (
    <motion.div
      initial={{y:-80, opacity:0}}
      animate={{y:0, opacity:1}}
      transition={{duration:.9, ease:[.22,1,.36,1]}}
      style={{
        position:"fixed", top:0, left:0, right:0,
        zIndex:200,
        display:"flex", justifyContent:"center",
        paddingTop:18,
        pointerEvents:"none",
      }}>

      {/* Pill */}
      <div ref={pillRef} style={{
        pointerEvents:"all",
        position:"relative",
        display:"flex", alignItems:"center", gap:2,
        padding:"5px 5px",
        borderRadius:100,
        background:"linear-gradient(108deg,rgba(174,183,132,0.42) 0%,rgba(140,158,96,0.30) 55%,rgba(174,183,132,0.38) 100%)",
        backdropFilter:"blur(28px) saturate(200%) hue-rotate(3deg)",
        WebkitBackdropFilter:"blur(28px) saturate(200%) hue-rotate(3deg)",
        border:"1.5px solid rgba(80,100,45,0.35)",
        boxShadow:`
          0 1.5px 0 rgba(220,235,185,0.50) inset,
          0 -1px 0 rgba(60,80,20,0.15) inset,
          0 12px 48px rgba(65,67,27,0.18),
          0 4px 12px rgba(65,67,27,0.10),
          0 0 0 1px rgba(255,255,255,0.12) inset
        `,
      }}>

        {/* ── Lamp — rendered once, positioned by measured lampX ── */}
        <AnimatePresence>
          {lampX !== null && NAV_LINKS.some(l=>l.id===displayActive) && (
            <motion.div
              key="lamp"
              initial={{opacity:0}}
              animate={{opacity:1, x: lampX}}
              exit={{opacity:0}}
              transition={{type:"spring", stiffness:300, damping:30}}
              style={{
                position:"absolute",
                top:-16,              // above the pill top border
                left:0,
                translateX:"-50%",   // centre on lampX
                pointerEvents:"none",
                zIndex:30,
                display:"flex",
                flexDirection:"column",
                alignItems:"center",
                width:0,             // zero-width so it doesn't push layout
              }}>
              {/* Bar */}
              <div style={{
                width:30, height:3, borderRadius:100,
                background:"rgba(50,65,12,0.95)",
                boxShadow:"0 0 10px rgba(80,110,20,0.8), 0 0 4px rgba(50,65,12,1)",
                flexShrink:0,
              }}/>
              {/* Outer glow cone */}
              <div style={{
                position:"absolute", top:3,
                width:80, height:36,
                background:"radial-gradient(ellipse 80% 100% at 50% 0%,rgba(174,183,132,0.55) 0%,transparent 85%)",
                filter:"blur(6px)",
              }}/>
              {/* Inner cone */}
              <div style={{
                position:"absolute", top:3,
                width:44, height:24,
                background:"radial-gradient(ellipse 80% 100% at 50% 0%,rgba(174,183,132,0.80) 0%,transparent 85%)",
                filter:"blur(2px)",
              }}/>
            </motion.div>
          )}
        </AnimatePresence>

        {/* Logo */}
        <motion.div whileHover={{scale:1.04}} whileTap={{scale:.96}}
          onClick={()=>go("hero")}
          style={{cursor:"pointer",userSelect:"none",
            padding:"6px 16px 6px 12px",
            borderRight:"1px solid rgba(80,100,45,0.18)",marginRight:4}}>
          <span className="serif" style={{fontSize:17,fontWeight:700,color:"#41431B",letterSpacing:"0.02em",whiteSpace:"nowrap"}}>
            AURELIUS<span style={{color:"#4A5C28",fontWeight:300}}>.</span>
          </span>
        </motion.div>

        {/* Links */}
        {NAV_LINKS.map(({label,id,external})=>{
          const isActive = displayActive===id;

          // Resume — styled as a distinct CTA pill, opens externally
          if (external) return (
            <div key={id}
              style={{marginLeft:4,paddingLeft:4,
                borderLeft:"1px solid rgba(80,100,45,0.18)"}}>
              <motion.a
                href={external}
                target="_blank"
                rel="noreferrer"
                whileHover={{scale:1.04, background:"rgba(65,67,27,0.88)"}}
                whileTap={{scale:.96}}
                style={{
                  display:"inline-flex", alignItems:"center", gap:7,
                  padding:"7px 18px", borderRadius:100,
                  background:"rgba(65,67,27,0.75)",
                  color:"#F8F3E1",
                  fontSize:12, fontWeight:500, letterSpacing:"0.08em",
                  textTransform:"uppercase",
                  fontFamily:"'DM Sans',sans-serif",
                  textDecoration:"none",
                  boxShadow:"0 2px 8px rgba(65,67,27,0.25)",
                  transition:"background .2s",
                  backdropFilter:"blur(8px)",
                  whiteSpace:"nowrap",
                }}>
                {/* Arrow-up-right icon */}
                <svg width="11" height="11" viewBox="0 0 11 11" fill="none">
                  <path d="M1 10L10 1M10 1H3M10 1V8" stroke="#F8F3E1" strokeWidth="1.6" strokeLinecap="round" strokeLinejoin="round"/>
                </svg>
                Resume
              </motion.a>
            </div>
          );

          // Regular scroll link
          return (
            <div key={id}
              ref={el=>{ linkRefs.current[id]=el; }}
              onClick={()=>go(id)}
              style={{position:"relative",cursor:"pointer"}}>

              <motion.span
                animate={{
                  opacity: isActive ? 1 : 0.38,
                  fontWeight: isActive ? 700 : 500,
                }}
                transition={{duration:.25}}
                style={{
                  position:"relative", display:"block",
                  fontSize:12, letterSpacing:"0.09em", textTransform:"uppercase",
                  color:"#41431B",
                  padding:"7px 18px", borderRadius:100,
                  userSelect:"none", zIndex:1,
                }}
                whileHover={{ opacity: isActive ? 1 : 0.72 }}
                onMouseEnter={e=>{ if(!isActive) e.currentTarget.style.letterSpacing="0.10em"; }}
                onMouseLeave={e=>{ if(!isActive) e.currentTarget.style.letterSpacing="0.09em"; }}>

                {/* Active pill — solid cream, strong shadow, no blur dependency */}
                <AnimatePresence>
                  {isActive && (
                    <motion.span
                      layoutId="active-pill"
                      initial={{opacity:0, scale:.9}}
                      animate={{opacity:1, scale:1}}
                      exit={{opacity:0, scale:.9}}
                      transition={{type:"spring", stiffness:380, damping:28}}
                      style={{
                        position:"absolute", inset:0, borderRadius:100, zIndex:-1,
                        background:"rgba(248,243,225,0.95)",
                        border:"1.5px solid rgba(65,67,27,0.18)",
                        boxShadow:"0 2px 12px rgba(65,67,27,0.18), 0 1px 0 rgba(255,255,255,0.8) inset",
                      }}/>
                  )}
                </AnimatePresence>

                {label}
              </motion.span>
            </div>
          );
        })}

      </div>
    </motion.div>
  );
};
